  function setPrevLabel(id,text){
    var el=q(id);if(!el)return;
    var p=el.previousElementSibling;
    if(p && p.classList && p.classList.contains('control-label'))p.textContent=text;
  }
  function setText(id,text){var el=q(id);if(el)el.textContent=text;}
  function apply(){
    var tabs={text:'✍ Format',print:'🖨 Print',comments:'💬 Comments',more:'🧰 More'};
    Object.keys(tabs).forEach(function(k){setText('wordTab-'+k,tabs[k]);});
    var current=document.querySelector('.word-tools-current label');if(current)current.textContent='Section';
    setPrevLabel('sf9FontFamilySelect','Font');
    setPrevLabel('sf9FontSizeSelect','Size');
    setPrevLabel('sf9SelectedTextSizeSelect','Text Size');
    setPrevLabel('sf9SelectedWeightSelect','Style');
    setPrevLabel('sf9SelectedSubjectIndentSelect','Position');
    setPrevLabel('printTermSelect','Term');
    setPrevLabel('paperSizeSelect','Paper');
    setPrevLabel('printLayoutSelect','Layout');
    setPrevLabel('attendancePrintSelect','Attendance');
    setPrevLabel('commentLanguageTop','Language');
    var z=document.querySelector('#wordAppearanceCard .zoom-controls');
    if(z){var zl=z.previousElementSibling;if(zl&&zl.classList.contains('control-label'))zl.textContent='Zoom';}
    setText('sf9SubjectIndentControlLabel','Subject');
    setText('sf9OuterBorderControlLabel','Border');
    var b=q('sf9TextSizeToolButton');if(b)b.textContent='✍ Select Text';
    b=q('sf9SubjectIndentToolButton');if(b)b.textContent='↔ Select Subject';
    b=q('sf9ResetFontButton');if(b)b.textContent='↶ Reset Font';
    b=q('resetHeaderTextButton');if(b)b.textContent='↶ Reset Headers';
    var pick=Array.from(document.querySelectorAll('#wordCommentsCard button')).find(function(x){return /Pick|Change|Edit Comment/i.test(x.textContent);});if(pick)pick.textContent='💬 Pick / Edit';
    var undo=Array.from(document.querySelectorAll('#wordCommentsCard button')).find(function(x){return /Undo/i.test(x.textContent);});if(undo)undo.textContent='↶ Undo';
    var pc=Array.from(document.querySelectorAll('#wordPrintActions button')).find(function(x){return /Current/i.test(x.textContent);});if(pc)pc.textContent='🖨 Print Current';
    var pa=Array.from(document.querySelectorAll('#wordPrintActions button')).find(function(x){return /All/i.test(x.textContent);});if(pa)pa.textContent='🗂 Print All';
    var seal=Array.from(document.querySelectorAll('#wordBrandCard button')).find(function(x){return /Seal/i.test(x.textContent);});if(seal)seal.textContent='🏛 DepEd Seal';
    var logo=Array.from(document.querySelectorAll('#wordBrandCard button')).find(function(x){return /Logo/i.test(x.textContent);});if(logo)logo.textContent='🏫 School Logo';
    var exp=Array.from(document.querySelectorAll('#wordDataCard button')).find(function(x){return /Export/i.test(x.textContent);});if(exp)exp.textContent='💾 Export';
    var res=Array.from(document.querySelectorAll('#wordDataCard button')).find(function(x){return /Restore/i.test(x.textContent);});if(res)res.textContent='📥 Restore';
    var resetSel=Array.from(document.querySelectorAll('#sf9SelectedSizeWrap button')).find(function(x){return /Reset/i.test(x.textContent);});if(resetSel)resetSel.textContent='↶ Reset Text';
    var resetPos=Array.from(document.querySelectorAll('#sf9SubjectIndentWrap button')).find(function(x){return /Reset/i.test(x.textContent);});if(resetPos)resetPos.textContent='↶ Reset Position';
    var note=q('sf9OuterBorderNote');if(note)note.textContent='Outer SF9 frame only. Inner table borders stay unchanged.';
  }
  function install(){
    [0,80,180,400,800].forEach(function(ms){setTimeout(apply,ms);});
    var oldTab=window.setWordToolTab;
    if(typeof oldTab==='function'&&!oldTab.__compactWrapped){
      window.setWordToolTab=function(){var r=oldTab.apply(this,arguments);setTimeout(apply,0);return r;};
      window.setWordToolTab.__compactWrapped=true;
    }
    var oldSetup=window.setupWordTools;
    if(typeof oldSetup==='function'&&!oldSetup.__compactWrapped){
      window.setupWordTools=function(){var r=oldSetup.apply(this,arguments);[50,150,350].forEach(function(ms){setTimeout(apply,ms);});return r;};
      window.setupWordTools.__compactWrapped=true;
    }
  }
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',install);else install();
})();
</script>

<script id="smartSf9CommentBankRestoreLogic">
(function(){
  function q(id){return document.getElementById(id);}

  window.openEditCurrentComment=function(){
    if(typeof window.openCommentPicker==='function')window.openCommentPicker();
    setTimeout(function(){
      var title=document.querySelector('#commentOverlay .modal-title');
      if(title)title.textContent='✏ Edit Teacher Comment';
      var ed=q('currentCommentEditor');
      if(ed){
        try{ed.scrollIntoView({behavior:'smooth',block:'center'});}catch(e){}
        ed.focus();
        try{ed.setSelectionRange(ed.value.length,ed.value.length);}catch(e){}
      }
    },100);
  };

  window.openFullCommentBank=function(){
    if(typeof window.openCommentPicker==='function')window.openCommentPicker();
    setTimeout(function(){
      var title=document.querySelector('#commentOverlay .modal-title');
      if(title)title.textContent='💬 Teacher Comment Bank';
      var search=q('commentSearch');
      if(search){try{search.scrollIntoView({behavior:'smooth',block:'center'});}catch(e){} search.focus();}
    },100);
  };

  function restoreCommentsTab(){
    var card=q('wordCommentsCard');if(!card)return;
    var p=card.querySelector('p');if(p)p.textContent='Open the comment bank or edit the current learner comment.';

    // Keep language first.
    var lang=q('commentLanguageTop');
    if(lang){
      var lab=lang.previousElementSibling;
      if(lab&&lab.classList&&lab.classList.contains('control-label')){lab.textContent='Language';card.appendChild(lab);}
      card.appendChild(lang);
    }

    // Reuse the original comment-picker button as Comment Bank.
    var bank=Array.from(document.querySelectorAll('.toolbar button,#wordCommentsCard button')).find(function(b){
      return String(b.getAttribute('onclick')||'').indexOf('openCommentPicker')>=0 || /Pick \/ Edit|Pick \/ Change|Comment Bank/i.test(b.textContent||'');
    });
    if(bank){
      bank.textContent='💬 Comment Bank';
      bank.setAttribute('onclick','openFullCommentBank()');
      bank.classList.add('comment-primary-btn');
      card.appendChild(bank);
    }

    var edit=q('wordEditCommentBtn');
    if(!edit){
      edit=document.createElement('button');
      edit.id='wordEditCommentBtn';
      edit.type='button';
      edit.className='comment-edit-btn';
      edit.textContent='✏ Edit Comment';
      edit.setAttribute('onclick','openEditCurrentComment()');
    }
    card.appendChild(edit);

    var helper=q('wordCommentHelper');
    if(!helper){
      helper=document.createElement('div');
      helper.id='wordCommentHelper';
      helper.className='comment-helper';
      helper.innerHTML='<b>Comment Bank</b> = choose/search saved comment options.<br><b>Edit Comment</b> = manually revise the current learner comment.';
    }
    card.appendChild(helper);

    var undo=Array.from(document.querySelectorAll('.toolbar button,#wordCommentsCard button')).find(function(b){
      return String(b.getAttribute('onclick')||'').indexOf('undoLastComment')>=0 || /^↶?\s*Undo/i.test(b.textContent||'');
    });
    if(undo){undo.textContent='↶ Undo';card.appendChild(undo);}

    var smooth=q('sf9SettingsCommentsSmooth');if(smooth)card.appendChild(smooth);
  }

  // Restore the default bank title every time the regular picker is opened directly.
  var originalOpen=window.openCommentPicker;
  if(typeof originalOpen==='function'&&!originalOpen.__commentBankTitleWrapped){
    window.openCommentPicker=function(){
      var r=originalOpen.apply(this,arguments);
      setTimeout(function(){var title=document.querySelector('#commentOverlay .modal-title');if(title)title.textContent='💬 Teacher Comment Bank';},0);
      return r;
    };
    window.openCommentPicker.__commentBankTitleWrapped=true;
  }

  function install(){
    [0,100,250,600].forEach(function(ms){setTimeout(restoreCommentsTab,ms);});
    var oldSetup=window.setupWordTools;
    if(typeof oldSetup==='function'&&!oldSetup.__commentBankRestored){
      window.setupWordTools=function(){var r=oldSetup.apply(this,arguments);[50,180,400].forEach(function(ms){setTimeout(restoreCommentsTab,ms);});return r;};
      window.setupWordTools.__commentBankRestored=true;
    }
    var oldTab=window.setWordToolTab;
    if(typeof oldTab==='function'&&!oldTab.__commentBankRestored){
      window.setWordToolTab=function(tab){var r=oldTab.apply(this,arguments);if(tab==='comments')setTimeout(restoreCommentsTab,0);return r;};
      window.setWordToolTab.__commentBankRestored=true;
    }
  }
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',install);else install();
})();
</script>
